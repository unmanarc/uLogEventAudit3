cmake_minimum_required(VERSION 3.16)
project(uAuditAnalyzer3 VERSION 3.0.0 DESCRIPTION "Unmanarc's Audit Analyzer" LANGUAGES CXX)
set(SVERSION ${PROJECT_VERSION})
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(Boost 1.75 REQUIRED COMPONENTS regex thread json)
include(cmake/Mantids30.cmake)
add_subdirectory(uanlz_alert)
add_subdirectory(uanlz_log2json)
add_subdirectory(uanlz_web)

include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
