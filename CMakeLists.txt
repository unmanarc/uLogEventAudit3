cmake_minimum_required(VERSION 3.16)
project(uLogEventAudit3 VERSION 3.0.0 DESCRIPTION "Unmanarc's Log and Event Audit Platform" LANGUAGES CXX)
set(SVERSION ${PROJECT_VERSION})
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(Boost 1.75 REQUIRED COMPONENTS regex thread json)
include(cmake/Mantids30.cmake)
add_library(audit_resource_sensing STATIC common/resource_sensing.cpp)
audit_target(audit_resource_sensing Mantids30_Core_Helpers Mantids30_Protocol_FastRPC1 Mantids30_DataFormat_JSON)
target_include_directories(audit_resource_sensing PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/common")
add_subdirectory(ul3_alert)
add_subdirectory(ul3_log2json)
add_subdirectory(ul3_web)

include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
